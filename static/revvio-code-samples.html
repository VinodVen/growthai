Sample 1: AI Integration — OpenAI API with Error Handling

What this does: Directly integrates OpenAI's GPT-4o-mini model to generate personalized marketing messages. Uses raw HTTP requests (not the OpenAI SDK) to avoid dependency conflicts in production — a deliberate architectural decision requiring advanced knowledge of Python HTTP clients and API design.

AI/ML Integration API Design Error Handling Production Architecture 
# main.py — OpenAI integration using requests library directly
# Decision: avoids httpx version conflict with openai SDK in production

import requests as _requests

_openai_api_key = os.getenv("OPENAI_API_KEY", "").strip()

def _call_openai(prompt, max_tokens=150, timeout=20):
    """
    Call OpenAI GPT-4o-mini via direct HTTP request.
    Returns generated text or raises Exception with error detail.
    """
    response = _requests.post(
        "https://api.openai.com/v1/chat/completions",
        headers={
            "Authorization": f"Bearer {_openai_api_key}",
            "Content-Type": "application/json"
        },
        json={
            "model": "gpt-4o-mini",
            "messages": [{"role": "user", "content": prompt}],
            "max_tokens": max_tokens,
            "temperature": 0.7
        },
        timeout=timeout
    )
    data = response.json()
    if "error" in data:
        raise Exception(data["error"]["message"])
    return data["choices"][0]["message"]["content"]

@app.route("/register", methods=["GET", "POST"])
def register():
    if request.method == "POST":
        email    = request.form.get("email", "").strip().lower()
        password = request.form.get("password", "")
        name     = request.form.get("business_name", "").strip()
        owner    = request.form.get("owner_name", "").strip()

        # Check for existing account — prevent duplicate registrations
        if Business.query.filter_by(email=email).first():
            flash("Email already registered.", "error")
            return redirect("/register")

        # Bcrypt hash — never store plaintext passwords
        hashed = bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()

        b = Business(
            business_name = name,
            owner_name    = owner,
            email         = email,
            password      = hashed,
            plan          = "free",
            trial_ends_at = datetime.utcnow() + timedelta(days=30),
        )
        b.slug = make_slug(name, "tmp")
        db.session.add(b)
        db.session.flush()
        b.slug = make_slug(name, b.id)
        db.session.commit()

        session["business_id"] = b.id
        session.permanent = True
        return redirect("/dashboard")

    return render_template("register.html")


@app.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        email    = request.form.get("email", "").strip().lower()
        password = request.form.get("password", "")
        b = Business.query.filter_by(email=email).first()

        # Constant-time comparison via bcrypt — prevents timing attacks
        if b and bcrypt.checkpw(password.encode(), b.password.encode()):
            session["business_id"] = b.id
            session.permanent = True
            b.last_login = datetime.utcnow()
            db.session.commit()
            return redirect("/dashboard")

        flash("Invalid email or password.", "error")
    return render_template("login.html")

@app.route("/stripe/webhook", methods=["POST"])
def stripe_webhook():
    payload = request.data
    sig_header = request.headers.get("Stripe-Signature")
    webhook_secret = os.getenv("STRIPE_WEBHOOK_SECRET")

    try:
        # Verify webhook signature — prevents spoofed payment events
        event = stripe.Webhook.construct_event(payload, sig_header, webhook_secret)
    except (ValueError, stripe.error.SignatureVerificationError):
        return jsonify({"error": "Invalid signature"}), 400

    if event["type"] == "checkout.session.completed":
        s = event["data"]["object"]
        b = Business.query.filter_by(stripe_customer_id=s.get("customer")).first()
        if b:
            sub = stripe.Subscription.retrieve(s["subscription"])
            price_id = sub["items"]["data"][0]["price"]["id"]
            b.plan = "pro" if price_id == STRIPE_PRICE_PRO else "starter"
            db.session.commit()

    elif event["type"] == "customer.subscription.deleted":
        s = event["data"]["object"]
        b = Business.query.filter_by(stripe_customer_id=s.get("customer")).first()
        if b:
            b.plan = "free"
            db.session.commit()

    return jsonify({"status": "ok"})

import random, string

@app.route("/api/do-checkin/<slug>", methods=["POST"])
def do_checkin(slug):
    b = Business.query.filter_by(slug=slug).first()
    if not b:
        return jsonify({"error": "Business not found"}), 404

    data       = request.get_json() or {}
    first_name = data.get("first_name", "").strip()
    last_name  = data.get("last_name", "").strip()
    phone      = data.get("phone", "").strip()
    email      = data.get("email", "").strip().lower()
    dob        = data.get("dob", "").strip()

    if not first_name or not phone:
        return jsonify({"error": "Name and phone required"}), 400

    # Find or create customer — prevent duplicates by phone
    customer = Customer.query.filter_by(
        business_id=b.id, phone=phone
    ).first()

    if not customer:
        customer = Customer(
            business_id   = b.id,
            first_name    = first_name,
            last_name     = last_name,
            email         = email,
            phone         = phone,
            dob           = dob,
            source        = "qr_checkin",
            sms_opted_in  = True,
            sms_opted_in_at = datetime.utcnow(),
            sms_opt_in_ip = request.remote_addr,
        )
        db.session.add(customer)
        db.session.flush()

    # Generate unique 10-char coupon code if welcome offer is enabled
    coupon_code = None
    coupon_data = None

    if getattr(b, 'welcome_offer_enabled', False):
        coupon_code = ''.join(
            random.choices(string.ascii_uppercase + string.digits, k=10)
        )
        expiry = datetime.utcnow() + timedelta(
            days=b.welcome_offer_expiry_days or 30
        )
        coupon_data = {
            "code":         coupon_code,
            "offer_amount": b.welcome_offer_amount or "$5",
            "offer_text":   b.welcome_offer_text or "off your next visit",
            "expiry_date":  expiry.strftime("%B %d, %Y"),
            "redeem_url":   url_for("redeem_page", code=coupon_code, _external=True),
        }

    # Log the check-in with coupon
    ci = CheckIn(
        business_id = b.id,
        customer_id = customer.id,
        name        = f"{first_name} {last_name}".strip(),
        phone       = phone,
        source      = "qr",
        coupon_code = coupon_code,
    )
    db.session.add(ci)
    db.session.commit()

    # Send SMS with coupon code via Twilio
    if coupon_code and twilio_client and phone:
        try:
            sms_body = (
                f"Welcome to {b.business_name}! 🎉\n"
                f"Your coupon: {coupon_code}\n"
                f"Get {b.welcome_offer_amount} {b.welcome_offer_text}.\n"
                f"Valid {b.welcome_offer_expiry_days} days."
            )
            twilio_client.messages.create(
                body=sms_body, from_=twilio_phone, to=phone
            )
        except Exception as e:
            print(f"SMS error: {e}")

    return jsonify({"success": True, "coupon": coupon_data})

def api_redeem_coupon(code):

    ci.coupon_redeemed_at = datetime.utcnow()
    db.session.commit()

    customer = Customer.query.get(ci.customer_id)
    return jsonify({
        "success": True,
        "name": customer.first_name if customer else ci.name
    })

Sample 6: Automated Marketing Engine (Cron Scheduler)

What this does: The automation engine runs every 30 minutes via an external cron trigger. It checks all businesses and fires AI-personalized messages for: birthday offers, win-back campaigns, weekly specials, and flash deals — each with AI-generated content via OpenAI, delivered via Twilio SMS. This is the core intelligence of the platform.

Automation Engine AI Personalization Cron Jobs Business Logic 
# main.py — Automation engine (simplified key logic)

@app.route("/cron/run-automations", methods=["GET"])
def run_automations():
    # Verify secret key — prevents unauthorized triggers
    secret = request.args.get("secret")
    if secret != os.getenv("CRON_SECRET"):
        return jsonify({"error": "Unauthorized"}), 403

    now = datetime.utcnow()
    results = {"birthday": 0, "winback": 0, "weekly": 0, "flash": 0}

    businesses = Business.query.all()
    for b in businesses:
        profile = BusinessProfile.query.filter_by(business_id=b.id).first()
        if not profile or not profile.setup_complete:
            continue

        customers = Customer.query.filter_by(
            business_id=b.id,
            unsubscribed=False
        ).all()

        # 1. Birthday automations
        if profile.auto_birthday:
            today_md = now.strftime("%m-%d")
            for c in customers:
                if c.dob and c.dob[5:] == today_md and c.phone:
                    prompt = (
                        f"Write a short birthday SMS for {c.first_name} "
                        f"from {b.business_name} ({profile.business_type}). "
                        f"Tone: {profile.tone}. Include a special birthday offer. "
                        f"Max 160 chars."
                    )
                    try:
                        msg = _call_openai(prompt, max_tokens=80, timeout=15)
                        _send_sms(c.phone, msg, b.id)
                        results["birthday"] += 1
                    except Exception as e:
                        print(f"Birthday SMS error: {e}")

        # 2. Win-back — customers not seen in 30+ days
        if profile.auto_winback:
            cutoff = now - timedelta(days=30)
            for c in customers:
                last = c.last_visit or c.created_at
                if last < cutoff and c.phone:
                    prompt = (
                        f"Write a win-back SMS for {c.first_name} who hasn't "
                        f"visited {b.business_name} in 30 days. "
                        f"Tone: {profile.tone}. Offer something special. "
                        f"Max 160 chars."
                    )
                    try:
                        msg = _call_openai(prompt, max_tokens=80, timeout=15)
                        _send_sms(c.phone, msg, b.id)
                        results["winback"] += 1
                    except Exception:
                        pass

        # 3. Weekly special — send every Tuesday
        if profile.auto_weekly and now.strftime("%A") == profile.weekly_send_day:
            last_sent = profile.last_weekly_summary
            if not last_sent or (now - last_sent).days >= 6:
                sms_customers = [c for c in customers if c.phone and c.sms_opted_in]
                if sms_customers:
                    prompt = (
                        f"Write a weekly special SMS for {b.business_name} "
                        f"({profile.business_type}). Tone: {profile.tone}. "
                        f"Max 160 chars. No placeholders."
                    )
                    try:
                        msg = _call_openai(prompt, max_tokens=80, timeout=15)
                        for c in sms_customers:
                            _send_sms(c.phone, msg, b.id)
                        profile.last_weekly_summary = now
                        results["weekly"] += len(sms_customers)
                    except Exception:
                        pass

    db.session.commit()
    return jsonify({"status": "done", "results": results, "time": str(now)})

Sample 7: Live Database Migration System

What this does: Handles schema migrations on a live production database without downtime. Uses ALTER TABLE IF NOT EXISTS patterns that are safe to run on every deploy. This is a critical production engineering skill — modifying a live PostgreSQL database without data loss or service interruption.

Database Engineering Zero-Downtime Migrations PostgreSQL Production Safety 
# main.py — Live database migration system
# Runs in background thread 2s after startup — safe for production

def _do_migrations():
    time.sleep(2)
    with app.app_context():
        try:
            db.create_all()  # create new tables

            # Add columns to existing tables without data loss
            # Each uses DO $$ ... END $$ for PostgreSQL-safe conditional add
            migrations = [
                # Business table — white-label branding columns
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS brand_name VARCHAR(200)",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS brand_color VARCHAR(20) DEFAULT '#7c3aed'",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS logo_url VARCHAR(500)",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS custom_subdomain VARCHAR(100)",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS disabled_features TEXT DEFAULT ''",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS onboarding_done BOOLEAN DEFAULT FALSE",
                # Welcome coupon system
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS welcome_offer_enabled BOOLEAN DEFAULT FALSE",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS welcome_offer_amount VARCHAR(50)",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS welcome_offer_text VARCHAR(200)",
                "ALTER TABLE businesses ADD COLUMN IF NOT EXISTS welcome_offer_expiry_days INTEGER DEFAULT 30",
                # Customer table — extended fields
                "ALTER TABLE customers ADD COLUMN IF NOT EXISTS first_name VARCHAR(120)",
                "ALTER TABLE customers ADD COLUMN IF NOT EXISTS last_name VARCHAR(120)",
                "ALTER TABLE customers ADD COLUMN IF NOT EXISTS dob VARCHAR(50)",
                "ALTER TABLE customers ADD COLUMN IF NOT EXISTS sms_opt_in_ip VARCHAR(60)",
                "ALTER TABLE customers ADD COLUMN IF NOT EXISTS loyalty_points INTEGER DEFAULT 0",
                # CheckIn table — coupon tracking
                "ALTER TABLE checkins ADD COLUMN IF NOT EXISTS coupon_code VARCHAR(20)",
                "ALTER TABLE checkins ADD COLUMN IF NOT EXISTS coupon_redeemed BOOLEAN DEFAULT FALSE",
                "ALTER TABLE checkins ADD COLUMN IF NOT EXISTS coupon_redeemed_at TIMESTAMP",
                "ALTER TABLE checkins ADD COLUMN IF NOT EXISTS customer_id INTEGER REFERENCES customers(id)",
            ]

            for sql in migrations:
                try:
                    db.session.execute(db.text(sql))
                except Exception as e:
                    print(f"Migration skipped: {e}")

            db.session.commit()
            print("Migrations complete.")

        except Exception as e:
            print(f"Migration error: {e}")
            db.session.rollback()


# Launch migrations in background so app starts immediately
threading.Thread(target=_do_migrations, daemon=True).start()
